defun fib-recursive(n):
  if n == 0:
    0
  elif n == 1:
    1
  else:
    fib-recursive(n - 1) + fib-recursive(n - 2)
  end
end

defun fib-iterative(n):
  loop :for i :below n
    :with f1 := 0
    :with f2 := 1
    :with tmp := nil
    :do tmp = f1 + f2
        f1 = f2
        f2 = tmp
    :finally return(f1)
  end
end

loop :for i :below 10
  :collect list(i, fib-iterative(i))
end
